import cv2
import os
import time
import keyboard
from cryptography.fernet import Fernet
# ---------- CONFIG ----------
PASSWORD = "mysecret"
MAX_ATTEMPTS = 3
IMAGE_FOLDER = ".intruder_logs"  # hidden folder (starts with dot)
LOCK_FOLDER = "files_to_lock"
KEY_FILE = "filekey.key"
if not os.path.exists(IMAGE_FOLDER):
    os.makedirs(IMAGE_FOLDER)
attempts = 0
typed_password = ""
# Encryption functions
def generate_key():
    key = Fernet.generate_key()
    with open(KEY_FILE, "wb") as key_file:
        key_file.write(key)
    return key

def encrypt_files(folder):
    key = generate_key()
    fernet = Fernet(key)
    for filename in os.listdir(folder):
        filepath = os.path.join(folder, filename)
        if os.path.isfile(filepath):
            with open(filepath, "rb") as file:
                data = file.read()
            encrypted = fernet.encrypt(data)
            with open(filepath, "wb") as file:
                file.write(encrypted)

# Capture image silently
def capture_image():
    cam = cv2.VideoCapture(0)
    time.sleep(0.4)  # short delay for focus
    ret, frame = cam.read()
    if ret:
        filename = os.path.join(IMAGE_FOLDER, f"intruder_{int(time.time())}.png")
        cv2.imwrite(filename, frame)
    cam.release()
    cv2.destroyAllWindows()

# Keyboard monitor
def on_key(event):
    global typed_password, attempts
    if event.name == "enter":  # user pressed Enter
        if typed_password == PASSWORD:
            typed_password = ""
        else:
            attempts += 1
            capture_image()
            typed_password = ""
            if attempts >= MAX_ATTEMPTS:
                encrypt_files(LOCK_FOLDER)
                os._exit(0)  # stop program
    elif event.name == "backspace":
        typed_password = typed_password[:-1]
    elif len(event.name) == 1:  # normal key
        typed_password += event.name

# Start background key listener
keyboard.on_press(on_key)

print("[Security] Stealth mode active...")
keyboard.wait()  # keeps running
